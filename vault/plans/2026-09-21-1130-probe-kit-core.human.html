Interfaces

  • probe.sh.list(repo: path): one line per row, ok or absent, throws exit 2
  • probe.sh.detect(repo: path): probe ids whose stack is present, throws exit 2
  • probe.sh.run(stage: string, repo: path, spec: path, max_cost: string, no_repo_code: flag, allow_repo_registry: flag, only: string): finding rows and exit code 0, 1 or 2, throws exit 2
  • probe.sh.diff(repo: path, base: string): finding rows in changed files, throws exit 2
  • probe.sh.scale(repo: path): files, lines and max-cost lines, throws exit 2
  • probe-registry.sh.probe_registry(repo: path, allow: int): registry lines with an origin column, throws exit 2
  • probe-registry.sh.probe_tool_present(run: string, trust: string, repo: path): exit code 0 or 1
  • probe-run.sh.probe_detect(cell: string, trust: string, repo: path): exit code 0 or 1
  • probe-run.sh.probe_run_row(origin: string, id: string, stack: string, stage: string, detect: string, run: string, parser: string, cost: string, trust: string, install: string): finding rows on stdout, throws status line
  • probe-run.sh.probe_exec(cwd: path, trust: string, cell: string, out: path, err: path): PROBE_RC
  • probe-run.sh.probe_abort(): -
  • probe-scope.sh.probe_files(repo: path, out: path): NUL-separated regular files
  • probe-scope.sh.probe_changed(repo: path, base: string, out: path): NUL-separated changed and untracked files, throws exit code 2
  • probe-scope.sh.probe_scale(repo: path, list: path): files, lines and max-cost lines
  • probe-scope.sh.probe_git(repo: path, args: string): git output
  • probe-scope.sh.probe_safe_file(repo: path, rel: string): exit code 0 or 1
  • probe-scope.sh.probe_normalize(rel: string): normalized path, throws exit code 1
  • probe-emit.sh.probe_finding(probe: string, file: path, line: int, severity: string, rule: string, message: string): one clean six-field row
  • probe-emit.sh.probe_sanitize(text: string): text without control bytes
  • probe-emit.sh.probe_args(argv: string): option variables set, throws exit 2
  • probe-emit.sh.probe_native_init(argv: string): repo, TMP and LIST set, throws exit 2
  • probe-emit.sh.probe_emit_sorted(check: string): sorted finding rows from stdin, throws exit code 1
  • probe-parsers.sh.parse_lizard_csv(id: string, repo: path): finding rows read from stdin
  • probe-parsers.sh.parse_typos_json(id: string, repo: path): finding rows read from stdin, throws exit code 3
  • probe-parsers.sh.parse_claude_validate(id: string, repo: path): finding rows read from stdin, throws exit code 3
  • probe-sql.sh.sql_facts(label: string, path: path): T, C and K fact lines
  • harness.sh.check(name: string, repo: path): finding rows, throws exit 2
  • sql-schema.sh.check(name: string, repo: path): finding rows, throws exit 2
  • similar-symbols.sh.check(repo: path, spec: path, names: string): finding rows, throws exit 2
